* Advanced Settings
** Functions
#+begin_src emacs-lisp
(defun shrink-left ()
    (interactive)
    (enlarge-window-horizontally 10))

(defun shrink-right ()
    (interactive)
    (shrink-window-horizontally 10))

(defun new-tab ()
    (interactive)
    (tab-new)
    (tab-bar-mode -1)
    (message "Added a new tab"))
#+end_src
