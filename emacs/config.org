* Advanced Settings
** Functions
#+begin_src emacs-lisp
(defun shrink-left ()
    (interactive)
    (enlarge-window-horizontally 10))

(defun shrink-right ()
    (interactive)
    (shrink-window-horizontally 10))

(defun consult-fd-everywhere ()
    (interactive)
    (let ((default-directory "~"))
        (consult-fd)))
#+end_src
